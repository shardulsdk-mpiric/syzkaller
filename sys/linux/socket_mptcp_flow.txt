# MPTCP (Multipath TCP, RFC 8684) MP_CAPABLE connection pair.
#
# syz_mptcp_pair_init() establishes a single MP_CAPABLE-negotiated MPTCP
# connection on loopback (client connect() + server accept(), both via
# IPPROTO_MPTCP) and threads it through a real resource, so later calls
# that operate on the pair have a genuine dependency edge on it -- see
# docs/pseudo_syscalls.md and the reproducibility discussion there.
#
# syz_mptcp_pair_close() is the consuming edge: it closes both fds and
# frees the pair.
#
# syz_mptcp_join_subflow() drives a real MP_JOIN handshake for a new
# subflow on an established pair, via the kernel's userspace
# path-manager netlink API (MPTCP_PM_CMD_SUBFLOW_CREATE) -- NORMAL mode
# only: the kernel computes a valid nonce/HMAC end-to-end, so this needs
# nothing beyond a stock CONFIG_MPTCP kernel. The resulting mptcp_subflow
# resource is the producer -> consumer edge this increment adds: it
# exists only because a join created it, and syz_mptcp_subflow_destroy()
# (MPTCP_PM_CMD_SUBFLOW_DESTROY) is the only call that consumes it -- the
# resource graph mirrors the real executor-side dependency, per the
# reproducibility discussion in docs/pseudo_syscalls.md.
#
# v0 scope (still, for this second increment): NORMAL mode only, no
# mutation, no nonce/HMAC capture, no kernel-side instrumentation --
# those land as a later increment. Runs on any stock CONFIG_MPTCP kernel.
#
# Two-netns pairs (MPTCP_INIT_PEER_NETNS): the client end stays in the executor's
# netns, the server end lives in a per-proc peer netns joined to it by a veth pair
# (10.0.0.1 <-> 10.0.0.2), so each end has its own in-kernel PM endpoint table and
# net.mptcp sysctls, as two hosts do: RM_ADDR and ADD_ADDR reach a peer with its own
# PM state. The ops that take a pair or an end work on either kind (a server end's
# commands go out from the peer netns); the in-kernel PM ops choose the netns
# structurally ($client / $server variants). Not yet on a two-netns pair: wire
# operations through the NFQUEUE layer (join_subflow's mut_op is not applied,
# subflow_stall and subflow_fail do nothing and fail; each prints a numbered "not
# applied" line on stderr), and listener_close_race (self-contained, always
# single-netns).

# Invalid default (0xff..ff -> -1 as the executor's long slot index, which its
# `slot < 0` guard rejects): a defaulted/orphaned handle must NOT alias live
# pool slot 0. Handles are real pool indices produced only by pair_init/join.
resource mptcp_pair[intptr]: 0xffffffffffffffff
resource mptcp_subflow[intptr]: 0xffffffffffffffff

# Path manager (PM) of each end. The kernel fixes an msk's PM when the msk is
# created (net.mptcp.path_manager, or pm_type on older kernels), and the two PMs
# exclude each other: userspace-PM commands reject a kernel-PM msk, and in-kernel
# PM configuration skips userspace-PM msks. The ends choose independently, as two
# real hosts do, so a pair can mix them. pair_init applies the choice only while it
# creates the pair and restores the netns setting afterwards, so nothing outlives
# the call.
#
# Each end that runs the userspace PM is also returned as a typed handle (same
# value as mptcp_pair, the pool slot). Ops that need a userspace PM on a given end
# take that end's handle, so they are not generated against a kernel-PM end they
# cannot act on. PM-agnostic ops keep taking mptcp_pair. A client end that runs
# the in-kernel PM is returned as mptcp_client_kernel, for the ops that need the
# in-kernel PM on the client.
resource mptcp_client_userspace[intptr]: 0xffffffffffffffff
resource mptcp_server_userspace[intptr]: 0xffffffffffffffff
resource mptcp_client_kernel[intptr]: 0xffffffffffffffff

# Harness constants for the pm argument (see .const + executor #define): bit 0
# selects the in-kernel PM for the client end, bit 1 for the server end.
define MPTCP_PAIR_PM_USERSPACE	0
define MPTCP_PAIR_PM_CLIENT_KERNEL	1
define MPTCP_PAIR_PM_SERVER_KERNEL	2
define MPTCP_PAIR_PM_KERNEL	3

syz_mptcp_pair_init$client_kernel(server_addr ptr[in, sockaddr_in], client_addr ptr[in, sockaddr_in], flags flags[mptcp_pair_init_flags], pm const[MPTCP_PAIR_PM_CLIENT_KERNEL], client ptr[out, mptcp_client_kernel], server ptr[out, mptcp_server_userspace]) mptcp_pair (remote_cover, timeout[2000])

# addr_id: local address id announced to the kernel for this subflow
# (id 0 is reserved/invalid on the wire -- the executor coerces it to 1).
# backup: sets MPTCP_PM_ADDR_FLAG_BACKUP on the new subflow when 1.
# mut_op: MPTCP_MUT_* -- corrupt the MP_JOIN on the egress path via the NFQUEUE
# mutation layer. NONE (0) leaves the join unmutated; HMAC_FLIP (1) flips a bit of
# the ACK's HMAC to exercise the kernel's MPJoinAckHMacFailure gate (the join then
# does not establish, by design). CRYPTO_SELFVERIFY (4) does NOT mutate: it is the
# ingress-injection Phase-0 spike -- it broadens the matcher to also observe the
# server SYN-ACK, captures R_A/R_B, and recomputes the ACK HMAC from the captured
# nonces + MP_CAPABLE keys, asserting it matches the kernel's (debug-only oracle).
# SELFVERIFY needs the pair created with MPTCP_INIT_CAPTURE_KEYS (else it reports
# no-keys). HMAC_FORGE (5) is ingress Phase 1: it OVERWRITES the client ACK HMAC
# with a VALID value recomputed from the captured nonces + MP_CAPABLE keys (the
# same value SELFVERIFY proved == the kernel's), so the subflow is ACCEPTED via a
# harness-authored HMAC that crossed subflow_hmac_valid -- reaching the
# subflow_syn_recv_sock ref-transfer region. Same-length rewrite. Like SELFVERIFY it
# needs MPTCP_INIT_CAPTURE_KEYS (else it DROPs the ACK, fail-loud -> no accept).
# On a two-netns pair mut_op is not applied yet (the join runs unmutated and prints a
# numbered "not applied" line on stderr): the engine hooks loopback egress, and
# THMAC_FLIP needs the server netns's egress.
# Harness-only constants (see .const + #define).
define MPTCP_MUT_NONE	0
define MPTCP_MUT_HMAC_FLIP	1
define MPTCP_MUT_HMAC_ZERO	2
define MPTCP_MUT_NONCE_FLIP	3
define MPTCP_MUT_CRYPTO_SELFVERIFY	4
define MPTCP_MUT_HMAC_FORGE	5
# Phase 2 (A1/B2): same-length DSS mapping corruption on egress data segments +
# SYN-ACK thmac flip. DSS_* reach the server mapping validation in softirq
# (subflow_check_data_avail -> validate_mapping / mptcp_validate_data_csum);
# THMAC_FLIP reaches client-side subflow_thmac_valid.
define MPTCP_MUT_DSS_SEQ	6
define MPTCP_MUT_DSS_SUBSEQ	7
define MPTCP_MUT_DSS_LEN_INF	8
define MPTCP_MUT_DSS_LEN_OFF	9
define MPTCP_MUT_DSS_CSUM	10
define MPTCP_MUT_DSS_DATA_FIN	11
define MPTCP_MUT_THMAC_FLIP	12
# Phase 3a (A5): length-growing graft of an unauthenticated ADD_ADDR echo option onto
# a client egress subflow segment -> mptcp_pm_add_addr_echoed. First op to exercise
# the NFQUEUE length-grow path (all prior ops are same-length).
define MPTCP_MUT_ADD_ADDR_ECHO	13
# Phase 3b (A5 authenticated): length-growing graft of an AUTHENTICATED non-echo
# ADD_ADDR (16 bytes) with a computed add-addr HMAC -> mptcp_pm_add_addr_received
# (the raced pm->remote single-slot). Needs MPTCP_INIT_CAPTURE_KEYS.
define MPTCP_MUT_ADD_ADDR_AUTH	14
# Phase 4 (A3): MP_FAIL graft on a live msk -> mptcp_pm_mp_fail_received (fallback
# machinery, the 006 cluster); to be raced vs teardown verbs.
define MPTCP_MUT_MP_FAIL	15
# Phase 4 (A4): MP_FASTCLOSE graft (rcvr_key echo) -> schedule_work teardown; teardown-vs-teardown race.
define MPTCP_MUT_MP_FASTCLOSE	16
# Phase 4: RM_ADDR + MP_PRIO grafts (PM addr churn / unlocked backup flip).
define MPTCP_MUT_RM_ADDR	17
define MPTCP_MUT_MP_PRIO	18
mptcp_mut_ops = MPTCP_MUT_NONE, MPTCP_MUT_HMAC_FLIP, MPTCP_MUT_HMAC_ZERO, MPTCP_MUT_NONCE_FLIP, MPTCP_MUT_CRYPTO_SELFVERIFY, MPTCP_MUT_HMAC_FORGE, MPTCP_MUT_DSS_SEQ, MPTCP_MUT_DSS_SUBSEQ, MPTCP_MUT_DSS_LEN_INF, MPTCP_MUT_DSS_LEN_OFF, MPTCP_MUT_DSS_CSUM, MPTCP_MUT_DSS_DATA_FIN, MPTCP_MUT_THMAC_FLIP, MPTCP_MUT_ADD_ADDR_ECHO, MPTCP_MUT_ADD_ADDR_AUTH, MPTCP_MUT_MP_FAIL, MPTCP_MUT_MP_FASTCLOSE, MPTCP_MUT_RM_ADDR, MPTCP_MUT_MP_PRIO
# The join is a userspace-PM command (SUBFLOW_CREATE) issued on the client msk, so it
# takes a userspace client end. mptcp_subflow is produced only here, so the ops that
# consume it inherit that guarantee.
#
# A mut_op other than NONE needs the executor's NFQUEUE layer, which needs kernel
# support (the nft queue expression, or the xtables NFQUEUE target with
# nft_compat); syz-manager disables this call on a kernel without it. The $plain
# variant never mutates and stays enabled there.
syz_mptcp_join_subflow(client mptcp_client_userspace, addr_id int8, backup int8[0:1], mut_op flags[mptcp_mut_ops]) mptcp_subflow (remote_cover)
syz_mptcp_join_subflow$plain(client mptcp_client_userspace, addr_id int8, backup int8[0:1], mut_op const[MPTCP_MUT_NONE]) mptcp_subflow (remote_cover)
syz_mptcp_subflow_destroy(subflow mptcp_subflow)

# Non-consuming op: takes a live mptcp_subflow WITHOUT destroying it (queries the
# parent msk via MPTCP_INFO). Unlike subflow_destroy, this does not free the
# handle, so the generator can sequence further ops on the same live subflow --
# the resource-model basis for adversarial multi-subflow lifecycle orderings.
syz_mptcp_subflow_info(subflow mptcp_subflow)

# Raw MP_JOIN SYN injection: sends one MP_JOIN SYN carrying the pair's server msk
# token to the loopback listener (raw socket). Drives the incoming-join-request
# path directly (cheaper than a full PM SUBFLOW_CREATE) and is the injection half
# of the close/teardown race class. Non-consuming; the pair stays usable.
syz_mptcp_inject_join_syn(pair mptcp_pair) (remote_cover)

# Teardown-race op: bursts MP_JOIN SYNs (server token) at the still-open listener
# while synchronously closing the accepted server msk (SO_LINGER), racing an
# injected SYN's softirq token lookup against the msk teardown -- the close/teardown
# race class. TERMINAL: consumes the pair (frees the slot) like pair_close. Explores
# for novel teardown UAF/NULL-deref bugs; if the race is won the kernel crashes.
syz_mptcp_join_close_race(pair mptcp_pair) (remote_cover)

# Lifecycle verb: close ONLY the accepted server msk (SO_LINGER, synchronous
# teardown) leaving the listener/client alive. Non-consuming. Purpose: give the
# generator a teardown verb to SEQUENCE and (via collide) run concurrently with
# joins, so the close/teardown race class emerges from syzkaller's own scheduling
# (the "breadth" path) rather than a hand-crafted timing op.
syz_mptcp_close_server(pair mptcp_pair) (remote_cover)

# Lifecycle verb: disconnect the accepted server msk via connect(AF_UNSPEC), driving
# mptcp_disconnect -> destroy_common (pm.ops NULL) -> pm_data_reset (pm.ops re-init)
# on a REUSED socket. Non-consuming (socket reused, not closed). Collided with a join
# (inject_join_syn), the ~50-line pm.ops NULL->reassign window is raceable -- the
# disconnect/reconnect reset-staleness surface. Reviewer-verified easier to hit than
# the close path.
syz_mptcp_disconnect(pair mptcp_pair) (remote_cover)

# Lifecycle verb (design §9d): listener-close / accept-vs-worker race. Self-contained
# (no pair resource) -- opens a fresh MPTCP listener, completes n_unaccepted MP_CAPABLE
# handshakes WITHOUT accept()ing them (they queue as unaccepted msks), sends a byte on
# each to schedule the msk worker, then closes the listener so inet_csk_listen_stop ->
# inet_child_forget frees the unaccepted children, racing the worker / softirq token
# lookup. Targets the unaccepted-teardown UAF class (0a3f4f1f / b6985b9b / 6aeed904).
# KASAN is the oracle.
syz_mptcp_listener_close_race(n_unaccepted int8[1:8]) (remote_cover)

# In-kernel path-manager configuration (netns-wide, like ip-mptcp(8)): endpoints and
# limits that the in-kernel PM acts on for every kernel-PM msk, both for pairs created
# afterwards and for live ones. Endpoint idx uses id idx+1 and address 127.0.1.(idx+1)
# on lo; the $initial variants use idx 8, id 9, on 127.0.0.1, the address of every
# pair's initial subflow (an endpoint on a host's primary address); the $auto
# variants use idx 16-23 on 127.0.4.(idx-15) and let the kernel choose the id, as
# ip mptcp endpoint add without "id" does (the executor reads the id back from a
# dump for later commands). The kernel hands out ids upwards from the highest one it
# has seen, so an auto endpoint added before the fixed ones takes their low ids and
# their adds fail with EBUSY, as they would for ip-mptcp(8) users. The executor's
# per-program netns reset flushes endpoints and restores the limits, in both netns.
# Both ends of a single-netns pair share the netns, so they share the endpoint table.
#
# Two-netns pairs: the $client_* / $server_* variants put the endpoint in the client
# end's netns (on veth-c) or in the server end's (on veth-s, the peer netns), each
# with its own table. The side is the add's last argument (where the bare variants
# say whether a signal endpoint gets a port; 1 still means a port), and the endpoint
# handle the add returns is that side bit ORed over the per-table handle above (fixed
# 0-7, initial 8, auto 16-23, id-0 256): 512 for $client, 1024 for $server, so
# $client fixed handles are 512-519, $server auto ones 1040-1047, and so on. The
# address blocks are the end's own: $client 10.0.2.(idx+1) fixed, 10.0.4.(idx-15)
# auto, 10.0.0.1 initial; $server 10.0.3.(idx+1), 10.0.5.(idx-15), 10.0.0.2. $client
# fixed and initial ids are offset by 32 (that table shares the executor's netns,
# and so its id space, with the loopback table: it is the same kernel table, so a
# bare flush also removes $client endpoints, and the auto-after-fixed id ordering
# applies across bare and $client). $client/$server ops build the peer
# netns and veth themselves when no two-netns pair did yet, so endpoints can exist
# before any pair, as on a real host. That first build (a netns, a veth pair, its
# addresses) takes ~150 ms on a KASAN kernel, more with several procs building at
# once. It happens once per executor proc, hence the timeout on the calls that can
# trigger it.
#
# Endpoints are resources, one family per side, so that removing or changing one is
# only generated for an endpoint that was added there, and changing to fullmesh only
# for a non-signal one (the kernel rejects fullmesh on a signal endpoint).
resource mptcp_pm_endpoint[intptr]: 0xffffffffffffffff
resource mptcp_pm_endpoint_subflow[mptcp_pm_endpoint]
resource mptcp_pm_endpoint_signal[mptcp_pm_endpoint]
resource mptcp_pm_endpoint_client[intptr]: 0xffffffffffffffff
resource mptcp_pm_endpoint_client_subflow[mptcp_pm_endpoint_client]
resource mptcp_pm_endpoint_client_signal[mptcp_pm_endpoint_client]
resource mptcp_pm_endpoint_server[intptr]: 0xffffffffffffffff
resource mptcp_pm_endpoint_server_subflow[mptcp_pm_endpoint_server]
resource mptcp_pm_endpoint_server_signal[mptcp_pm_endpoint_server]

# Scenario ops: conditions the kernel acts on that no sequence of the other ops
# creates. Each is bounded in time and leaves nothing behind.
#
# syz_mptcp_subflow_stall: one subflow of a pair stops making progress for hold_ms
# while data is pushed both ways, then is released. target picks the subflow the
# handle names or the pair's initial one; mode DELAY makes each of its packets
# arrive delay_ms late (its RTT grows, under TCP's minimum RTO), BLOCK holds its
# packets until the end, DROP drops them; dirs picks the direction(s). The other
# subflows run at full speed and go idle between pushes: the state the scheduler
# and the reinjection logic react to. Done in the NFQUEUE layer, scoped to the
# subflow's addresses and ports, so other pairs of the process are not affected.
# Like the mutating join, it needs NFQUEUE kernel support (see join_subflow).
define MPTCP_STALL_TARGET_SUBFLOW	0
define MPTCP_STALL_TARGET_INITIAL	1
define MPTCP_STALL_DELAY	0
define MPTCP_STALL_BLOCK	1
define MPTCP_STALL_DROP	2
define MPTCP_STALL_TO_SERVER	1
define MPTCP_STALL_TO_CLIENT	2
mptcp_stall_targets = MPTCP_STALL_TARGET_SUBFLOW, MPTCP_STALL_TARGET_INITIAL
mptcp_stall_modes = MPTCP_STALL_DELAY, MPTCP_STALL_BLOCK, MPTCP_STALL_DROP
mptcp_stall_dirs = MPTCP_STALL_TO_SERVER, MPTCP_STALL_TO_CLIENT
syz_mptcp_subflow_stall(subflow mptcp_subflow, target flags[mptcp_stall_targets], mode flags[mptcp_stall_modes], dirs flags[mptcp_stall_dirs], delay_ms int8[1:100], hold_ms int16[0:1000], data ptr[in, array[int8]], data_len len[data]) (timeout[1500], prog_timeout[2000], remote_cover)
