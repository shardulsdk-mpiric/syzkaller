# In-kernel QUIC (net/quic, Xin Long / lxin, RFC 9000) Phase-1 connection carrier.
#
# syz_quic_conn_init() brings up an IPPROTO_QUIC (261) socket pair on loopback
# in the executor's netns and drives it to ESTABLISHED using FAKE per-level
# crypto secrets -- no real TLS. It mirrors the kernel's OWN selftest recipe
# (tools/testing/selftests/net/quic_test.c: set_fake_keys()/do_handshake()):
# the kernel does not parse the TLS messages carried in CRYPTO frames of
# long-header packets (unless ALPN demux is on), so fake handshake bytes reach
# established deterministically. This answers the gate Xin Long himself named --
# "most paths cannot be reached because they require the handshake to complete"
# -- and is the analog of syz_mptcp_pair_init. The established pair is held in a
# small static pool; the slot index is returned as the quic_conn resource.
#
# syz_quic_conn_close() closes both fds and frees the slot (consuming edge).
#
# syz_quic_path_challenge() CROSSES a capture-echo gate: it triggers connection
# migration on one end (QUIC_SOCKOPT_CONNECTION_MIGRATION) so the kernel emits a
# PATH_CHALLENGE carrying 8 bytes of entropy generated by get_random_bytes() --
# a value that is NOT obtainable through any socket API, only off the wire. A
# NFQUEUE wire engine (reused from the MPTCP carrier, UDP variant) captures that
# entropy from the plaintext 1-RTT frame (disable_1rtt_encryption makes 1-RTT
# packets plaintext) and echoes it back in the peer's PATH_RESPONSE so the
# kernel's path-validation check (net/quic/frame.c crypto_memneq) succeeds and
# migration completes. mode QUIC_PC_ECHO_CAPTURED echoes the captured entropy
# (validation PASSES); mode QUIC_PC_ECHO_WRONG echoes a corrupted value (the
# negative control -- validation REJECTS), confirming the gate is real.

# Invalid default (0xff..ff -> -1 as the executor's long slot index, rejected by
# its `slot < 0` guard): a defaulted/orphaned handle must not alias live slot 0.
resource quic_conn[intptr]: 0xffffffffffffffff

# Harness-only constants (no kernel header defines these for syz-extract): give
# the values here so `make extract` stays reproducible instead of failing with
# "defined for none of the arches". Must match the executor #defines + the .const.
# Crypto cipher types mirror TLS_CIPHER_* (uapi/linux/tls.h); 0 defaults to GCM128.
define QUIC_CIPHER_DEFAULT	0
define QUIC_CIPHER_AES_GCM_128	51
define QUIC_CIPHER_AES_GCM_256	52
define QUIC_CIPHER_AES_CCM_128	53
define QUIC_CIPHER_CHACHA20_POLY1305	54
quic_cipher = QUIC_CIPHER_DEFAULT, QUIC_CIPHER_AES_GCM_128, QUIC_CIPHER_AES_GCM_256, QUIC_CIPHER_AES_CCM_128, QUIC_CIPHER_CHACHA20_POLY1305

# Path-challenge echo mode. ECHO_CAPTURED (0): echo the entropy captured off the
# wire -> kernel path validation PASSES. ECHO_WRONG (1): echo a corrupted value
# -> validation REJECTS (negative control proving the gate is not bypassed).
define QUIC_PC_ECHO_CAPTURED	0
define QUIC_PC_ECHO_WRONG	1
quic_pc_modes = QUIC_PC_ECHO_CAPTURED, QUIC_PC_ECHO_WRONG

quic_conn_opts {
# AF_INET / AF_INET6 (loopback). Phase 1 verifies AF_INET.
	family		flags[quic_conn_family, int32]
# disable_1rtt: negotiate disable_1rtt_encryption pre-handshake so short-header
# 1-RTT packets are plaintext on the wire. Required (=1) for syz_quic_path_challenge
# to capture/echo the PATH_CHALLENGE entropy; harmless otherwise.
	disable_1rtt	int8[0:1]
	cipher		flags[quic_cipher, int32]
}

quic_conn_family = AF_INET, AF_INET6

# Out-param snapshot of the established connection, harvested from the pair
# (mirrors syz_mptcp_subflow_info's out-param shape). state: 1 = established.
quic_conn_state {
	state		int32
	cipher		int32
	client_port	int16
	server_port	int16
	active_estabs	int32
	passive_estabs	int32
}

syz_quic_conn_init(opts ptr[in, quic_conn_opts], state ptr[out, quic_conn_state]) quic_conn (remote_cover)
syz_quic_conn_close(conn quic_conn)
syz_quic_path_challenge(conn quic_conn, mode flags[quic_pc_modes]) (remote_cover)
